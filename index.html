โค้ด Google Apps Script (รองรับทั้ง doGet และ doPost) 
const SHEET_ID = '1QISYTMlOtz7o5gJ9_zHw-MizBYKtHTnyIWdwrSUc59E';
const SHEET_NAME = 'Charge_Log';
const HEADERS = [
  'id', 'date', 'monthKey', 'startSoc', 'endSoc', 'addedPct',
  'netKwh', 'grossKwh', 'tariffMode', 'cost', 'co2Saved', 'score', 'note'
];

function getSheet_() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sh = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
  if (sh.getLastRow() === 0) sh.appendRow(HEADERS);
  return sh;
}

function jsonOut_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  try {
    const sh = getSheet_();
    const lastRow = sh.getLastRow();
    const lastCol = sh.getLastColumn();
    if (lastRow <= 1) return jsonOut_({ ok: true, logs: [] });
    const headers = sh.getRange(1, 1, 1, lastCol).getValues()[0];
    const rows = sh.getRange(2, 1, lastRow - 1, lastCol).getValues();
    const logs = rows
      .filter(r => r.some(c => c !== '' && c !== null))
      .map((r, idx) => {
        const item = {};
        headers.forEach((h, i) => {
          if (!h) return;
          let val = r[i];
          if (h === 'date' && Object.prototype.toString.call(val) === '[object Date]' && !isNaN(val)) {
            val = Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd');
          }
          item[h] = val;
        });
        if (!item.id) item.id = 'row-' + (idx + 2);
        if (!item.monthKey && item.date) item.monthKey = String(item.date).slice(0, 7);
        return item;
      });
    return jsonOut_({ ok: true, logs });
  } catch (err) {
    return jsonOut_({ ok: false, error: err.message });
  }
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    const sh = getSheet_();
    const data = e && e.postData && e.postData.contents ? JSON.parse(e.postData.contents) : {};

    if (data.action === 'delete' && data.id) {
      const lastRow = sh.getLastRow();
      if (lastRow > 1) {
        const ids = sh.getRange(2, 1, lastRow - 1, 1).getValues().flat();
        const foundIdx = ids.findIndex(v => String(v) === String(data.id));
        if (foundIdx !== -1) sh.deleteRow(foundIdx + 2);
      }
      return jsonOut_({ ok: true, deletedId: data.id });
    }

    if (data.action === 'clearAll') {
      const lastRow = sh.getLastRow();
      if (lastRow > 1) sh.deleteRows(2, lastRow - 1);
      return jsonOut_({ ok: true, cleared: true });
    }

    const row = HEADERS.map(k => data[k] !== undefined ? data[k] : '');
    sh.appendRow(row);
    return jsonOut_({ ok: true, id: data.id || '' });
  } catch (err) {
    return jsonOut_({ ok: false, error: err.message });
  } finally {
    lock.releaseLock();
  }
}
