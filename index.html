โค้ด Apps Script ด้านล่างเขียนข้อมูลลงคอลัมน์ A ถึง L (12 ช่อง) ของแท็บ Charge_Log ตรงหัวตารางเป๊ะๆ (ไม่เยื้อง)

โค้ด Google Apps Script (เขียนลง A:L ตรงช่อง) 
const SHEET_ID = '1QISYTMlOtz7o5gJ9_zHw-MizBYKtHTnyIWdwrSUc59E';
const SHEET_NAME = 'Charge_Log';

function getTargetSheet_() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) throw new Error('ไม่พบแท็บชื่อ ' + SHEET_NAME);
  return sheet;
}

function jsonResponse_(payload) {
  return ContentService
    .createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}

function findFirstEmptyLogRow_(sheet) {
  const startRow = 5;
  const maxCheck = Math.max(sheet.getLastRow() + 1, 200);
  const colA = sheet.getRange(startRow, 1, maxCheck - startRow + 1, 1).getValues();
  for (let i = 0; i < colA.length; i++) {
    if (colA[i][0] === '' || colA[i][0] === null) return startRow + i;
  }
  return maxCheck + 1;
}

function doGet(e) {
  try {
    const sheet = getTargetSheet_();
    const lastRow = sheet.getLastRow();
    if (lastRow < 5) return jsonResponse_({ ok: true, logs: [] });
    const rows = sheet.getRange(5, 1, lastRow - 4, 12).getValues();
    const logs = [];
    rows.forEach((r, idx) => {
      const rowNum = idx + 5;
      if (!r[0] && !r[1] && !r[2]) return;
      const dateStr = Object.prototype.toString.call(r[0]) === '[object Date]' && !isNaN(r[0])
        ? Utilities.formatDate(r[0], Session.getScriptTimeZone(), 'yyyy-MM-dd')
        : String(r[0] || '').slice(0, 10);
      logs.push({
        id: 'row-' + rowNum,
        rowNum,
        date: dateStr,
        startSoc: Number(r[1]) || 0,
        endSoc: Number(r[2]) || 0,
        addedPct: r[3] !== '' ? Number(r[3]) : Math.max(0, (Number(r[2]) || 0) - (Number(r[1]) || 0)),
        netKwh: Number(r[4]) || 0,
        grossKwh: Number(r[5]) || 0,
        tariffMode: String(r[6] || 'Flat'),
        cost: Number(r[7]) || 0,
        co2Saved: Number(r[8]) || 0,
        score: Number(r[9]) || 85,
        monthKey: String(r[10] || dateStr.slice(0, 7)),
        note: String(r[11] || '')
      });
    });
    return jsonResponse_({ ok: true, logs });
  } catch (err) {
    return jsonResponse_({ ok: false, error: err.message });
  }
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    const sheet = getTargetSheet_();
    const body = e && e.postData && e.postData.contents ? JSON.parse(e.postData.contents) : {};
    if (body.action === 'delete' && body.rowNum && Number(body.rowNum) >= 5) {
      sheet.getRange(Number(body.rowNum), 1, 1, 12).clearContent();
      return jsonResponse_({ ok: true, deleted: true });
    }
    const targetRow = findFirstEmptyLogRow_(sheet);
    const dateVal = body.date ? new Date(body.date + 'T00:00:00') : new Date();
    sheet.getRange(targetRow, 1, 1, 12).setValues([[
      dateVal,
      Number(body.startSoc) || 0,
      Number(body.endSoc) || 0,
      Number(body.addedPct) || 0,
      Number(body.netKwh) || 0,
      Number(body.grossKwh) || 0,
      String(body.tariffMode || 'Flat'),
      Number(body.cost) || 0,
      Number(body.co2Saved) || 0,
      Number(body.score) || 85,
      String(body.monthKey || String(body.date || '').slice(0, 7)),
      String(body.note || '')
    ]]);
    return jsonResponse_({ ok: true, rowNum: targetRow });
  } catch (err) {
    return jsonResponse_({ ok: false, error: err.message });
  } finally {
    lock.releaseLock();
  }
}
